\section*{\color{nttdark}\sffamily Guía rápida}
\begin{tcolorbox}[colback=nttlight,colframe=nttline,boxrule=.4pt,arc=2mm]
  \textbf{Objetivo.} Cargar informes PDF de tasación, extraer sus campos,
  revisar las alertas y descargar un Excel con los casos validados.
\end{tcolorbox}

\begin{tabular}{@{}p{.29\linewidth}p{.64\linewidth}@{}}
\color{nttblue}\textbf{Acceso al Sheet} & \href{https://docs.google.com/spreadsheets/d/1eOUVKs2gtmO1ligTwHVwdMiuTX37vzmL0LRbI3xeMMM/edit?gid=1416849096#gid=1416849096}{Abrir Google Sheet de Tasaciones}.\\[0.45cm]
\color{nttblue}\textbf{Regla de seguridad} & Utilice solo PDF autorizados. No comparta información de clientes ni exporte filas con observaciones pendientes.\\[0.45cm]
\color{nttblue}\textbf{Salida} & El Excel descargado no conserva macros. Para el paso NT3270/HIPO, use la plantilla corporativa \texttt{.xlsm} aprobada.
\end{tabular}

\paso{01}{Abrir el Sheet}{
Abra el enlace de la aplicación. La pestaña \textbf{Panel de Entrada} es el punto de inicio del lote.
}{01}
\pasoinicio{Después de abrir el Sheet, ubique el botón \textbf{Automatizar transacciones}. Este botón abre el menú que inicia el flujo.}
\paso{03}{Iniciar la carga}{
En el menú \textbf{Automatización de Tasaciones}, seleccione \textbf{Cargar PDFs}.
}{02}
\paso{04}{Seleccionar los PDF}{
Seleccione los informes PDF correspondientes al lote. Verifique los nombres antes de continuar.
}{03}
\paso{05}{Confirmar la selección}{
Revise la lista mostrada en el panel lateral y pulse \textbf{Cargar lote}.
}{04}
\paso{06}{Esperar la carga}{
Espere el mensaje final. Los archivos se guardan en Drive y el Panel de Entrada se actualiza.
}{05}
\paso{07}{Procesar el lote}{
Abra el menú y seleccione \textbf{Procesar y clasificar archivos}.
}{06}
\paso{08}{Confirmar el procesamiento}{
Si se solicita confirmación, acepte para iniciar la extracción de datos.
}{07}
\paso{09}{Esperar el resultado}{
No cierre el Sheet durante el análisis. El sistema informa cuando el lote termina.
}{08}
\paso{10}{Leer el resumen}{
El aviso muestra cuántos casos quedaron listos y cuántos requieren revisión. Pulse \textbf{Aceptar}.
}{09}
\paso{11}{Revisar casos listos}{
Abra \textbf{Listos para el Banco} y compruebe que los campos necesarios estén completos.
}{10}
\paso{12}{Abrir la bandeja de revisión}{
Los casos con datos ausentes o inconsistentes aparecen en \textbf{Bandeja de Revisión}.
}{11}
\paso{13}{Abrir el panel de revisión}{
Seleccione \textbf{Abrir panel de revisión}. El panel muestra un caso pendiente por vez.
}{12}
\paso{14}{Validar la información}{
Compruebe el PDF original y corrija únicamente datos verificables.
}{13}
\paso{15}{Aprobar y continuar}{
Pulse \textbf{Aplicar Corrección y Siguiente}. El caso pasa a \textbf{Listos para el Banco}. Luego use \textbf{Generar Excel de casos validados}.
}{14}
